184:class SMRIDataset(Dataset):
185:
186:    def __init__(
187:        self,
188:        dataframe,
189:        image_dir,
190:        augment=False,
191:    ):
192:
193:        self.dataframe = (
194:            dataframe
195:            .reset_index(drop=True)
196:            .copy()
197:        )
198:
199:        self.image_dir = Path(
200:            image_dir
201:        )
202:
203:        self.augment = bool(
204:            augment
205:        )
206:
207:    def __len__(self):
208:
209:        return len(
210:            self.dataframe
211:        )
212:
213:    def __getitem__(self, index):
214:
215:        row = self.dataframe.iloc[
216:            index
217:        ]
218:
219:        subject_id = str(
220:            row["subject_id"]
221:        )
222:
223:        label = float(
224:            row["label"]
225:        )
226:
227:        image_path = (
228:            self.image_dir
229:            / f"{subject_id}.npy"
230:        )
231:
232:        if not image_path.exists():
233:
234:            raise FileNotFoundError(
235:                f"Tensor not found: "
236:                f"{image_path}"
237:            )

239:        image = np.load(

241:        ).astype(
242:            np.float32,
243:            copy=False,
244:        )
245:
246:        if tuple(image.shape) != (
247:            EXPECTED_TENSOR_SHAPE

250:            raise RuntimeError(
251:                f"Unexpected tensor shape "

284:            image

305:                f"augmentation for "
306:                f"{subject_id}"

310:            image
311:        )

313:        label_tensor = torch.tensor(
314:            [label],

413:# ============================================================
414:# LOSS
415:# ============================================================
416:
417:def create_criterion(
418:    dataframe,
419:    device,
420:):
421:
422:    labels = (
423:        dataframe[
424:            "label"
425:        ]
426:        .astype(int)
427:        .to_numpy()
428:    )
429:
430:    negative_count = int(
431:        np.sum(
432:            labels == 0
433:        )
434:    )
435:
436:    positive_count = int(
437:        np.sum(
438:            labels == 1
439:        )
440:    )
441:
442:    if (
443:        negative_count == 0
444:        or positive_count == 0
445:    ):
446:
447:        raise RuntimeError(
448:            "Training partition does not "
449:            "contain both classes."
450:        )
451:
452:    pos_weight_value = (
453:        negative_count
454:        / positive_count
455:    )
456:
457:    pos_weight = torch.tensor(
458:        [pos_weight_value],
459:        dtype=torch.float32,
460:        device=device,
461:    )
462:
463:    criterion = (
464:        nn.BCEWithLogitsLoss(
465:            pos_weight=pos_weight
466:        )
467:    )
468:
469:    return (
470:        criterion,
471:        pos_weight_value,
472:    )

475:# ============================================================
476:# TRAIN ONE EPOCH
477:# ============================================================
478:
479:def train_one_epoch(
480:    model,
481:    loader,
482:    criterion,
483:    optimizer,
484:    device,
485:):
486:
487:    model.train()
488:
489:    running_loss = 0.0
490:
491:    labels_all = []
492:
493:    probabilities_all = []
494:
495:    subject_count = 0
496:
497:    for (
498:        images,
499:        labels,
500:        _,
501:    ) in loader:
502:
503:        images = images.to(
504:            device,
505:            non_blocking=True,
506:        )
507:
508:        labels = labels.to(
509:            device,
510:            non_blocking=True,
511:        )
512:
513:        optimizer.zero_grad(
514:            set_to_none=True
515:        )
516:
517:        logits = model(
518:            images
519:        )
520:
521:        loss = criterion(
522:            logits,
523:            labels,
524:        )
525:
526:        loss.backward()
527:
528:        optimizer.step()
529:
530:        batch_size = images.size(
531:            0
532:        )
533:
534:        running_loss += (
535:            loss.item()
536:            * batch_size
537:        )
538:
539:        subject_count += (
540:            batch_size
541:        )
542:
543:        probabilities = (
544:            torch.sigmoid(
545:                logits
546:            )
547:        )
548:
549:        labels_all.extend(
550:            labels.detach()
551:            .cpu()
552:            .numpy()
553:            .reshape(-1)
554:            .tolist()
555:        )
556:
557:        probabilities_all.extend(
558:            probabilities.detach()
559:            .cpu()
560:            .numpy()
561:            .reshape(-1)
562:            .tolist()
563:        )
564:
565:    if subject_count == 0:
566:
567:        raise RuntimeError(
568:            "Training loader is empty."
569:        )
570:
571:    epoch_loss = (
572:        running_loss
573:        / subject_count
574:    )
575:
576:    metrics = calculate_metrics(
577:        labels_all,
578:        probabilities_all,
579:    )
580:
581:    return (
582:        epoch_loss,
583:        metrics,
584:    )

984:    epochs_without_improvement = 0
